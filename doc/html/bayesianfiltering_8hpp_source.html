136 Result MultivariateZeroMeanGaussianSampler(Eigen::VectorXd & sample, Eigen::MatrixXd & covariance_matrix)
137 {
138 std::random_device rd;
139 std::mt19937 gen(rd());
140 std::normal_distribution<double> dist(0.0, 1.0);
141
142 // Verify dimensions
143 if (covariance_matrix.rows() != covariance_matrix.cols()) return ResultFormat(false, "covariance_matrix should be square (%li x %li)",covariance_matrix.rows(),covariance_matrix.cols());
144
145 // Compute Cholesky Decomposition (LLT)
146 Eigen::LLT<Eigen::MatrixXd> llt(covariance_matrix);
147 Eigen::MatrixXd L = llt.matrixL();
148
149 // Generate independent standard normal vector Z
150 Eigen::VectorXd Z = Eigen::VectorXd::NullaryExpr(covariance_matrix.rows(), [&]() { return dist(gen); });
151
152 // Transform to Multivariate Normal
153 sample = L * Z;
154
155 return ResultFormat(true,"");
156 }

165 Result MultivariateGaussianSampler(Eigen::VectorXd & sample, Eigen::VectorXd & mean_vector, Eigen::MatrixXd & covariance_matrix)
166 {
167 // Verify dimensions
168 if (mean_vector.rows() != covariance_matrix.rows()) return ResultFormat(false, "mean_vector number of lines (%li) different of covariance_matrix number of lines (%li)",mean_vector.rows(),covariance_matrix.rows());
169 // Covariance Matrix will be checked at zero mean Gaussian sampler
170
171 // Generate sample from zero mean sample:
172 Result res = MultivariateZeroMeanGaussianSampler(sample, covariance_matrix);
173 if (!res.first) return res;
174
175 sample += mean_vector;
176
177 return res;
178 }

704
705 // Innovation
706 Y_predicted = linear_model.C * X;
707 V_predicted = Y - Y_predicted;
708 S_predicted = linear_model.C * P * linear_model.C.transpose() + linear_model.R;
709 d2_predicted = (V_predicted.transpose() * S_predicted.inverse() * V_predicted)(0, 0);
710
711 // State update
712 Eigen::MatrixXd G = P * linear_model.C.transpose() * S_predicted.inverse();
713 X = X + G * V_predicted;
714 //P = (Eigen::MatrixXd::Identity(X.rows(), X.rows()) - G * linear_model.C) * P;
715 // More numericaly stable P update:
716 P = (Eigen::MatrixXd::Identity(X.rows(), X.rows()) - G * linear_model.C) * P *((Eigen::MatrixXd::Identity(X.rows(), X.rows()) - G * linear_model.C).transpose()) + G*linear_model.R*G.transpose();
717

719 Y_corrected = linear_model.C * X;
720 V_corrected = Y - Y_corrected;
721 S_corrected = linear_model.C * P * linear_model.C.transpose() + linear_model.R;
722 d2_corrected = (V_corrected.transpose() * S_corrected.inverse() * V_corrected)(0, 0);

726

729

849 void InitializeWeights(Eigen::VectorXd & Wsigma, double & kappa, int Nx)
850 {
851 int Nsigma = 2*Nx + 1;
852
853 Wsigma.resize(Nsigma,1);
854
855 kappa = (3 > Nx-3) ? 3 : Nx-3;
856 Wsigma(0) = kappa/(Nx+kappa);
857 for (int i = 1; i < Wsigma.rows(); ++i) {
858 Wsigma(i) = 1/(2.0*(Nx+kappa));
859 }
860 Wsigma = Wsigma / Wsigma.sum(); // Garante soma 1.

882 Result Prediction(Eigen::VectorXd & U, double t, StateSpaceNonlinearModelBase& nonlinear_model) override
883 {

886 // Prediction
887 // The function x(k) = f(x(k-1),u(k),w(k),t(k)) is converted to x(k) = f(xa(k),u(k),t(k)) with xa(k) = [x(k-1); w(k)]
888 int Nx = nonlinear_model.Nx;
889 int Nw = nonlinear_model.Nw;
890 int DimX = Nx + Nw;
891 int DimY = Nx;
892
893 Eigen::VectorXd Wsigma;
894 double kappa;
895 InitializeWeights(Wsigma, kappa, DimX);
896 int Nsigma = Wsigma.rows();
897
898 // Generate augmented Xa and Pa
899 Eigen::VectorXd Xa(DimX);
900 Xa << X, Eigen::VectorXd::Zero(Nw);
901
