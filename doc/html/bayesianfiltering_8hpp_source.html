136 Result MultivariateZeroMeanGaussianSampler(Eigen::VectorXd & sample, Eigen::MatrixXd & covariance_matrix)
137 {
138 std::random_device rd;
139 std::mt19937 gen(rd());
140 std::normal_distribution<double> dist(0.0, 1.0);
141
142 // Verify dimensions
143 if (covariance_matrix.rows() != covariance_matrix.cols()) return ResultFormat(false, "covariance_matrix should be square (%li x %li)",covariance_matrix.rows(),covariance_matrix.cols());
144
145 // Compute Cholesky Decomposition (LLT)
146 Eigen::LLT<Eigen::MatrixXd> llt(covariance_matrix);
147 Eigen::MatrixXd L = llt.matrixL();
148
149 // Generate independent standard normal vector Z
150 Eigen::VectorXd Z = Eigen::VectorXd::NullaryExpr(covariance_matrix.rows(), [&]() { return dist(gen); });
151
152 // Transform to Multivariate Normal
153 sample = L * Z;
154
155 return ResultFormat(true,"");
156 }

165 Result MultivariateGaussianSampler(Eigen::VectorXd & sample, Eigen::VectorXd & mean_vector, Eigen::MatrixXd & covariance_matrix)
166 {
167 // Verify dimensions
168 if (mean_vector.rows() != covariance_matrix.rows()) return ResultFormat(false, "mean_vector number of lines (%li) different of covariance_matrix number of lines (%li)",mean_vector.rows(),covariance_matrix.rows());
169 // Covariance Matrix will be checked at zero mean Gaussian sampler
170
171 // Generate sample from zero mean sample:
172 Result res = MultivariateZeroMeanGaussianSampler(sample, covariance_matrix);
173 if (!res.first) return res;
174
175 sample += mean_vector;
176
177 return res;
178 }

571 Eigen::MatrixXd Gi = G * V_predicted;
572 for(int i=0; i < Gi.rows(); ++i){
